# TEST site (v2, beacon branch) - copied from the production server block of arbiters.verdikta.org.
# Serves the test checkout /root/verdikta/applications-test (never the live one) and the test API on :6008.
# Install: see ops/test-sites/README.md. certbot --nginx adds the 443/SSL block.
server {
    server_name test.arbiters.verdikta.org;

    # Defense-in-depth hardening (block hidden-file probes, /@fs, common scanner paths)
    include /etc/nginx/snippets/verdikta-hardening.conf;

    # Client — served as static files from the Vite build output.
    root /root/verdikta/applications-test/example-arbiters/client/dist-test;
    index index.html;

    # Test site: keep it out of search engines.
    add_header X-Robots-Tag "noindex, nofollow" always;
    location = /robots.txt {
        default_type text/plain;
        return 200 "User-agent: *\nDisallow: /\n";
    }

    # Optional: password-protect the whole test site.
    # auth_basic "Verdikta test";
    # auth_basic_user_file /etc/nginx/verdikta-test.htpasswd;

    location ^~ /assets/ {
        access_log off;
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
        try_files $uri =404;
    }

    location ~* \.[a-z0-9]+$ {
        try_files $uri =404;
    }

    location / {
        try_files $uri $uri/ /index.html;
    }

    location = /index.html {
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
    }

    # Health check (served by Express)
    location = /health {
        proxy_pass http://127.0.0.1:6008/health;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # API (served by Express)
    location /api/ {
        client_max_body_size 25m;
        proxy_pass http://127.0.0.1:6008/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen 80;
}
