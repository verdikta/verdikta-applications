# TEST site (v2, beacon branch) - copied from the production server block of bounties-testnet.verdikta.org.
# Serves the test checkout /root/verdikta/applications-test (never the live one) and the test API on :6006.
# Install: see ops/test-sites/README.md. certbot --nginx adds the 443/SSL block.
server {
    server_name test.bounties-testnet.verdikta.org;

    # Defense-in-depth hardening (block hidden-file probes, /@fs, common scanner paths)
    include /etc/nginx/snippets/verdikta-hardening.conf;

    # Testnet client (Base Sepolia) — served as static files from Vite build.
    # `vite build --watch` keeps dist-base-sepolia/ up-to-date as source changes.
    root /root/verdikta/applications-test/example-bounty-program/client/dist-test-base-sepolia;
    index index.html;

    # Test site: keep it out of search engines.
    add_header X-Robots-Tag "noindex, nofollow" always;
    location = /robots.txt {
        default_type text/plain;
        return 200 "User-agent: *\nDisallow: /\n";
    }

    # Optional: password-protect the whole test site.
    # auth_basic "Verdikta test";
    # auth_basic_user_file /etc/nginx/verdikta-test.htpasswd;

    location ^~ /assets/ {
        access_log off;
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
        try_files $uri =404;
    }

    location ~* \.[a-z0-9]+$ {
        try_files $uri =404;
    }

    location / {
        try_files $uri $uri/ /index.html;
    }

    location = /index.html {
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
    }

    # Agent discovery + feed routes (served by Express, not the SPA)
    location = /agents.txt {
        proxy_pass http://localhost:6006/agents.txt;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /feed.xml {
        proxy_pass http://localhost:6006/feed.xml;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /llms.txt {
        proxy_pass http://localhost:6006/llms.txt;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /sitemap.xml {
        proxy_pass http://localhost:6006/sitemap.xml;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /health {
        proxy_pass http://localhost:6006/health;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Server-rendered share/receipt pages + OG card images. These MUST reach the
    # backend, not the static SPA. ^~ so the "ends in .ext" regex location below
    # does not grab /og/<...>.png before this prefix match.
    location ^~ /r/ {
        proxy_pass http://localhost:6006/r/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location ^~ /og/ {
        proxy_pass http://localhost:6006/og/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Testnet API (Base Sepolia)
    location ^~ /api/ {
        client_max_body_size 25m;
        proxy_pass http://localhost:6006/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen 80;
}
