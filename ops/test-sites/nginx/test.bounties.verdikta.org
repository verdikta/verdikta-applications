# TEST site (v2, beacon branch) - copied from the production server block of bounties.verdikta.org.
# Serves the test checkout /root/verdikta/applications-test (never the live one) and the test API on :6005.
# Install: see ops/test-sites/README.md. certbot --nginx adds the 443/SSL block.
server {
    server_name test.bounties.verdikta.org;

    # Defense-in-depth hardening (block hidden-file probes, /@fs, common scanner paths)
    include /etc/nginx/snippets/verdikta-hardening.conf;

    # Mainnet client (Base) — served as static files from the Vite build output.
    # `vite build --watch` keeps dist-base/ up-to-date as source files change.
    root /root/verdikta/applications-test/example-bounty-program/client/dist-test-base;
    index index.html;

    # Test site: keep it out of search engines.
    add_header X-Robots-Tag "noindex, nofollow" always;
    location = /robots.txt {
        default_type text/plain;
        return 200 "User-agent: *\nDisallow: /\n";
    }

    # Optional: password-protect the whole test site.
    # auth_basic "Verdikta test";
    # auth_basic_user_file /etc/nginx/verdikta-test.htpasswd;

    # Long-cache the hashed Vite assets (filenames include content hashes).
    location ^~ /assets/ {
        access_log off;
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
        try_files $uri =404;
    }

    # Anything with a file extension must exist on disk — prevents the SPA
    # fallback from masking 404s on /foo.bak, /config.yml, etc.
    location ~* \.[a-z0-9]+$ {
        try_files $uri =404;
    }

    # SPA fallback for clean (extensionless) routes — React Router paths.
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Never cache the HTML entrypoint
    location = /index.html {
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header X-Robots-Tag "noindex, nofollow" always;
        include /etc/nginx/snippets/verdikta-security-headers.conf;
    }

    # Agent discovery + feed routes (served by Express, not the SPA)
    location = /agents.txt {
        proxy_pass http://127.0.0.1:6005/agents.txt;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /feed.xml {
        proxy_pass http://127.0.0.1:6005/feed.xml;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /llms.txt {
        proxy_pass http://127.0.0.1:6005/llms.txt;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /sitemap.xml {
        proxy_pass http://127.0.0.1:6005/sitemap.xml;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location = /health {
        proxy_pass http://127.0.0.1:6005/health;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Server-rendered share/receipt pages + OG card images. These MUST reach the
    # backend, not the static SPA. ^~ so the "ends in .ext" regex location below
    # does not grab /og/<...>.png before this prefix match.
    location ^~ /r/ {
        proxy_pass http://127.0.0.1:6005/r/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location ^~ /og/ {
        proxy_pass http://127.0.0.1:6005/og/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Mainnet API (Base)
    location ^~ /api/ {
        client_max_body_size 25m;
        proxy_pass http://127.0.0.1:6005/api/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    listen 80;
}
